### 監控指令
```bash
python3 -c "
import serial, time
ser = serial.Serial('/dev/ttyACM1', 115200, timeout=5)
time.sleep(2)
ser.flushInput()
for i in range(30):
    if ser.in_waiting > 0:
        data = ser.read(ser.in_waiting)
        print(data.decode('utf-8', errors='replace'))
        break
    time.sleep(0.5)
ser.close()
"
```
